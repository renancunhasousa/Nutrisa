import React, { useState, useEffect, useRef } from 'react';
import { initializeApp } from "firebase/app";
import { getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken } from "firebase/auth";
import { getFirestore, doc, setDoc, getDoc, collection, onSnapshot, deleteDoc, addDoc, updateDoc, query } from "firebase/firestore";
import { 
  FileText, 
  Upload, 
  Stethoscope, 
  User, 
  Settings, 
  ArrowRight, 
  CheckCircle, 
  Copy, 
  Download, 
  RefreshCw,
  AlertCircle,
  FileOutput,
  List,
  Eye, 
  EyeOff,
  BrainCircuit,
  Sparkles,
  Save,
  FolderOpen,
  Trash2,
  Plus,
  MessageCircle,
  HelpCircle,
  X,
  Cloud,
  CloudOff,
  Key,
  Lock,
  Cpu,
  Coins
} from 'lucide-react';

// --- CONFIGURAÇÃO DO FIREBASE ---
let db, auth, appId;
try {
  const firebaseConfig = JSON.parse(__firebase_config);
  const app = initializeApp(firebaseConfig);
  auth = getAuth(app);
  db = getFirestore(app);
  appId = typeof __app_id !== 'undefined' ? __app_id : 'medi-compile-ai';
} catch (e) {
  console.error("Erro ao inicializar Firebase.", e);
}

const App = () => {
  // Estados de Autenticação e Configurações
  const [user, setUser] = useState(null);
  const [openaiKey, setOpenaiKey] = useState('');
  const [openaiModel, setOpenaiModel] = useState('gpt-4o-mini');
  const [showKeyInput, setShowKeyInput] = useState(false);
  const [isSavingKey, setIsSavingKey] = useState(false);

  // Estados de Sistema
  const [isSavingDraft, setIsSavingDraft] = useState(false);
  const [dbStatus, setDbStatus] = useState('connecting');
  
  // Estados de Dados
  const [patientData, setPatientData] = useState('');
  const [doctorData, setDoctorData] = useState('');
  const [template, setTemplate] = useState('');
  const [savedTemplates, setSavedTemplates] = useState([]);
  const [selectedTemplateId, setSelectedTemplateId] = useState('');
  const [templateNameInput, setTemplateNameInput] = useState('');
  const [result, setResult] = useState('');
  const [isGenerating, setIsGenerating] = useState(false);
  const [usageCost, setUsageCost] = useState(null); 
  
  // Interface
  const [activeTab, setActiveTab] = useState('input');
  const [notification, setNotification] = useState(null);
  const [pdfLibLoaded, setPdfLibLoaded] = useState(false);
  const [detectedVars, setDetectedVars] = useState([]);
  const [extraContent, setExtraContent] = useState(null); 
  const [isExtraLoading, setIsExtraLoading] = useState(false);

  // --- PREÇOS (por 1M de tokens) ---
  const PRICING = {
    'gpt-4o-mini': { input: 0.15, output: 0.60 },
    'gpt-5-mini': { input: 0.25, output: 2.00 },
    'gpt-5.4-mini': { input: 0.75, output: 4.50 }
  };
  const BRL_RATE = 5.20; 

  // --- FUNÇÕES DE NOTIFICAÇÃO ---
  const showNotification = (msg, type) => {
    setNotification({ msg, type });
    setTimeout(() => setNotification(null), 3000);
  };

  const copyToClipboard = (text) => {
    const el = document.createElement('textarea');
    el.value = text;
    document.body.appendChild(el);
    el.select();
    try {
      document.execCommand('copy');
      showNotification("Copiado com sucesso!", "success");
    } catch (err) {
      showNotification("Erro ao copiar.", "error");
    }
    document.body.removeChild(el);
  };

  // --- PERSISTÊNCIA DE DADOS (FIREBASE) ---
  useEffect(() => {
    if (!auth) return;
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (error) {
        setDbStatus('error');
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (u) => {
      setUser(u);
      if (u) setDbStatus('connected');
    });
    return () => unsubscribe();
  }, []);

  useEffect(() => {
    if (!user || !db) return;
    const loadUserData = async () => {
      try {
        const configRef = doc(db, 'artifacts', appId, 'users', user.uid, 'settings', 'openai');
        const configSnap = await getDoc(configRef);
        if (configSnap.exists()) {
          const data = configSnap.data();
          setOpenaiKey(data.key || '');
          setOpenaiModel(data.model || 'gpt-4o-mini');
        }
        const draftRef = doc(db, 'artifacts', appId, 'users', user.uid, 'app_data', 'draft');
        const draftSnap = await getDoc(draftRef);
        if (draftSnap.exists()) setTemplate(draftSnap.data().content || '');
      } catch (e) { console.error("Erro ao carregar dados", e); }
    };
    loadUserData();
  }, [user]);

  useEffect(() => {
    if (!user || !db || !template) return;
    const timeoutId = setTimeout(async () => {
      setIsSavingDraft(true);
      try {
        const draftRef = doc(db, 'artifacts', appId, 'users', user.uid, 'app_data', 'draft');
        await setDoc(draftRef, { content: template }, { merge: true });
      } catch (e) { }
      finally { setIsSavingDraft(false); }
    }, 2000);
    return () => clearTimeout(timeoutId);
  }, [template, user]);

  useEffect(() => {
    if (!user || !db) return;
    const templatesRef = collection(db, 'artifacts', appId, 'users', user.uid, 'templates');
    const unsubscribe = onSnapshot(templatesRef, (snapshot) => {
      const temps = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      temps.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
      setSavedTemplates(temps);
    });
    return () => unsubscribe();
  }, [user]);

  // --- INTEGRAÇÃO OPENAI ---
  const callOpenAI = async (prompt, isJson = false) => {
    if (!openaiKey) {
      setShowKeyInput(true);
      showNotification("Por favor, configure a chave da OpenAI.", "error");
      throw new Error("Chave não configurada.");
    }
    const body = {
      model: openaiModel,
      messages: [
        { role: "system", content: "Você é um assistente médico especialista em análise de prontuários." },
        { role: "user", content: prompt }
      ],
      temperature: 0.3
    };
    if (isJson) body.response_format = { type: "json_object" };
    try {
      const response = await fetch("https://api.openai.com/v1/chat/completions", {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${openaiKey}`
        },
        body: JSON.stringify(body)
      });
      if (!response.ok) {
        const errData = await response.json();
        throw new Error(errData.error?.message || `Erro ${response.status}`);
      }
      const data = await response.json();
      const { prompt_tokens, completion_tokens } = data.usage;
      const modelPricing = PRICING[openaiModel] || PRICING['gpt-4o-mini'];
      const costUSD = (prompt_tokens * modelPricing.input / 1000000) + (completion_tokens * modelPricing.output / 1000000);
      setUsageCost({
        usd: costUSD.toFixed(4),
        brl: (costUSD * BRL_RATE).toFixed(4)
      });
      return data.choices[0].message.content;
    } catch (error) {
      console.error("Erro OpenAI:", error);
      throw error;
    }
  };

  const handleSaveOpenAIConfig = async () => {
    if (!user || !db) return;
    setIsSavingKey(true);
    try {
      const configRef = doc(db, 'artifacts', appId, 'users', user.uid, 'settings', 'openai');
      await setDoc(configRef, { key: openaiKey, model: openaiModel }, { merge: true });
      showNotification("Configurações atualizadas!", "success");
      setShowKeyInput(false);
    } catch (e) {
      showNotification("Erro ao guardar ajustes.", "error");
    } finally {
      setIsSavingKey(false);
    }
  };

  // --- LÓGICA DE GERAÇÃO ---
  const generateReport = async () => {
    if (!openaiKey) return setShowKeyInput(true);
    setIsGenerating(true);
    setExtraContent(null);
    setUsageCost(null);
    setActiveTab('result'); 
    try {
      let finalReport = template;
      const tags = template.match(/{{\s*[\wÀ-ÿ]+\s*}}/g) || [];
      if (tags.length === 0) {
        setResult(template);
        setIsGenerating(false);
        return;
      }
      const variableNames = [...new Set(tags.map(v => v.replace(/{{|}}/g, '').trim()))];
      const prompt = `Extraia as variáveis: ${variableNames.join(', ')} dos seguintes textos e retorne JSON.
        TEXTO PACIENTE: ${patientData}
        TEXTO MÉDICO: ${doctorData}
        Use "Não informado" se ausente.`;
      const responseText = await callOpenAI(prompt, true);
      const aiData = JSON.parse(responseText);
      variableNames.forEach(name => {
        const regex = new RegExp(`{{\\s*${name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\s*}}`, 'g');
        const val = aiData[name] || aiData[name.toLowerCase()] || aiData[name.toUpperCase()] || "Não informado";
        finalReport = finalReport.replace(regex, String(val));
      });
      setResult(finalReport);
      showNotification(`Relatório gerado com sucesso!`, "success");
    } catch (error) { 
      showNotification(error.message, "error");
      setResult(`Erro: ${error.message}`);
    } finally { setIsGenerating(false); }
  };

  const handleExtraAnalysis = async (type) => {
    if (!result || !openaiKey) return;
    setIsExtraLoading(true);
    setExtraContent(null);
    try {
      const prompt = type === 'patient' 
        ? `Explique este relatório médico de forma simples e acolhedora para o paciente: ${result}`
        : `Indique lacunas e perguntas de seguimento clínico baseadas nestes dados: ${patientData} ${doctorData}`;
      const text = await callOpenAI(prompt);
      setExtraContent({ type, text });
      showNotification("Análise concluída!", "success");
    } catch (e) { showNotification("Erro na análise secundária.", "error"); }
    finally { setIsExtraLoading(false); }
  };

  // --- PDF ---
  useEffect(() => {
    const script = document.createElement('script');
    script.src = "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js";
    script.async = true;
    script.onload = () => {
      if (window.pdfjsLib) {
        window.pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js`;
        setPdfLibLoaded(true);
      }
    };
    document.body.appendChild(script);
  }, []);

  useEffect(() => {
    const matches = template.match(/{{\s*[\wÀ-ÿ]+\s*}}/g) || [];
    setDetectedVars([...new Set(matches)]);
  }, [template]);

  return (
    <div className="min-h-screen bg-slate-50 font-sans text-slate-800 antialiased">
      <style>{`
        .custom-scrollbar::-webkit-scrollbar {
          width: 6px;
        }
        .custom-scrollbar::-webkit-scrollbar-track {
          background: transparent;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb {
          background: #e2e8f0;
          border-radius: 10px;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover {
          background: #cbd5e1;
        }
      `}</style>
      <header className="bg-indigo-900 text-white p-5 shadow-xl relative">
        <div className="max-w-6xl mx-auto flex items-center justify-between relative z-10">
          <div className="flex items-center gap-4">
            <div className="bg-white/10 p-2 rounded-xl text-teal-300 border border-white/10 shadow-inner"><BrainCircuit size={28} /></div>
            <div>
              <h1 className="text-xl md:text-2xl font-black tracking-tight">Anamnese AI</h1>
              <p className="text-indigo-300 text-[10px] font-bold uppercase tracking-widest">Powered by: Renan Sousa</p>
            </div>
          </div>
          <div className="flex items-center gap-2">
            <div className="hidden sm:flex items-center px-3 py-1.5 rounded-lg bg-black/30 border border-white/5 text-[9px] font-black uppercase tracking-tighter text-indigo-200 gap-2 mr-1">
              <Cpu size={12} className="text-teal-400" />
              {openaiModel}
            </div>
            <button onClick={() => setShowKeyInput(!showKeyInput)} className={`p-2.5 rounded-xl transition-all border ${openaiKey ? 'border-teal-500/50 text-teal-400 bg-teal-500/10' : 'border-amber-500/50 text-amber-400 bg-amber-500/10'} hover:scale-105`} title="Configurar OpenAI API"><Key size={18} /></button>
            <div className={`hidden md:flex items-center gap-2 text-[9px] font-black uppercase px-3 py-1.5 rounded-full border bg-black/20 ${dbStatus === 'connected' ? 'border-green-500/30 text-green-400' : 'border-red-500/30 text-red-400'}`}>
              <div className={`w-1.5 h-1.5 rounded-full ${dbStatus === 'connected' ? 'bg-green-500' : 'bg-red-500'}`} />
              {dbStatus === 'connected' ? "Nuvem" : "Offline"}
            </div>
            <button onClick={() => {setPatientData(''); setDoctorData(''); setResult(''); setActiveTab('input'); setUsageCost(null);}} className="text-[10px] font-black uppercase bg-white/10 hover:bg-white/20 px-4 py-2 rounded-lg border border-white/10 flex items-center gap-2 transition-all"><RefreshCw size={14} /> Novo</button>
          </div>
        </div>
        {showKeyInput && (
          <div className="absolute top-full left-0 w-full bg-indigo-950 p-6 border-t border-white/10 z-[60] animate-fadeIn shadow-2xl">
            <div className="max-w-4xl mx-auto">
              <div className="flex justify-between items-center mb-6"><h3 className="text-xs font-black uppercase tracking-[0.2em] text-teal-400 flex items-center gap-2"><Cpu size={14}/> Inteligência Artificial</h3><button onClick={() => setShowKeyInput(false)} className="text-white/40 hover:text-white"><X size={18}/></button></div>
              <div className="flex flex-col md:flex-row gap-6 items-start">
                <div className="flex-1 w-full space-y-2"><label className="text-[9px] font-black text-indigo-300 uppercase ml-2">OpenAI API Key</label>
                  <div className="relative"><Key className="absolute left-4 top-1/2 -translate-y-1/2 text-white/30" size={16} /><input type="password" value={openaiKey} onChange={(e) => setOpenaiKey(e.target.value)} placeholder="sk-..." className="w-full bg-black/40 border border-white/10 rounded-2xl py-3 pl-12 pr-4 text-sm font-mono focus:border-teal-500 outline-none transition-all text-teal-50" /></div>
                </div>
                <div className="w-full md:w-80 space-y-2"><label className="text-[9px] font-black text-indigo-300 uppercase ml-2">Modelo Selecionado</label>
                  <div className="relative"><Cpu className="absolute left-4 top-1/2 -translate-y-1/2 text-white/30" size={16} /><select value={openaiModel} onChange={(e) => setOpenaiModel(e.target.value)} className="w-full bg-black/40 border border-white/10 rounded-2xl py-3 pl-12 pr-4 text-sm focus:border-teal-500 outline-none transition-all text-white appearance-none cursor-pointer font-bold"><option value="gpt-4o-mini">GPT-4o mini (Mais rápido)</option><option value="gpt-5-mini">GPT-5 mini (Intermediário)</option><option value="gpt-5.4-mini">GPT-5.4 mini (Mais inteligente)</option></select></div>
                </div>
                <div className="w-full md:w-auto pt-5"><button onClick={handleSaveOpenAIConfig} disabled={isSavingKey} className="w-full md:w-auto bg-teal-600 hover:bg-teal-500 px-8 py-3 rounded-2xl text-xs font-black uppercase tracking-widest disabled:opacity-50 transition-all flex items-center justify-center gap-2 shadow-lg">{isSavingKey ? <RefreshCw size={14} className="animate-spin" /> : <Save size={16} />} Gravar Ajustes</button></div>
              </div>
            </div>
          </div>
        )}
      </header>

      <main className="max-w-6xl mx-auto p-4 md:p-8">
        <div className="flex gap-1 mb-8 border-b border-slate-200 overflow-x-auto scrollbar-hide">
          {['input', 'template', 'result'].map(tab => (
            <button key={tab} onClick={() => setActiveTab(tab)} className={`px-8 py-4 font-black text-[10px] uppercase tracking-[0.15em] rounded-t-2xl transition-all whitespace-nowrap ${activeTab === tab ? 'bg-white text-indigo-900 border-x border-t border-slate-200 shadow-sm' : 'text-slate-400 hover:text-slate-500'}`}>
              {tab === 'input' ? '1. Coleta de Dados' : tab === 'template' ? '2. Estrutura Modelo' : '3. Relatório IA'}
            </button>
          ))}
        </div>

        {activeTab === 'input' && (
          <div className="grid md:grid-cols-2 gap-8 animate-fadeIn">
            {[
              { title: 'Questionário do Paciente', icon: <User size={18}/>, val: patientData, set: setPatientData, ph: "Introduza aqui o questionário respondido pelo paciente ou carregue um PDF..." },
              { title: 'Notas Clínicas / Anamnese', icon: <Stethoscope size={18}/>, val: doctorData, set: setDoctorData, ph: "Notas da consulta, exame físico e observações clínicas..." }
            ].map((field, idx) => (
              <div key={idx} className="bg-white rounded-3xl shadow-sm border border-slate-200 p-6 flex flex-col h-[550px]">
                <div className="flex justify-between items-center mb-5 text-indigo-800 font-black uppercase tracking-tighter">
                  <div className="flex items-center gap-3">{field.icon} <h2 className="text-sm">{field.title}</h2></div>
                  <label className="cursor-pointer text-[10px] font-black bg-slate-100 hover:bg-slate-200 text-slate-600 px-4 py-2 rounded-xl border border-slate-200 transition-all shadow-sm">
                    <Upload size={12} className="inline mr-2" /> PDF / TXT
                    <input type="file" accept=".txt, .pdf" className="hidden" onChange={(e) => {
                       const file = e.target.files[0];
                       if (!file) return;
                       if (file.type === 'application/pdf') {
                         const reader = new FileReader();
                         reader.onload = async (ev) => {
                           const arrayBuffer = ev.target.result;
                           const pdf = await window.pdfjsLib.getDocument({ data: arrayBuffer }).promise;
                           let fullText = "";
                           for (let i = 1; i <= pdf.numPages; i++) {
                             const page = await pdf.getPage(i);
                             const content = await page.getTextContent();
                             fullText += content.items.map(item => item.str).join(' ') + "\n";
                           }
                           field.set(fullText);
                           showNotification("PDF lido!", "success");
                         };
                         reader.readAsArrayBuffer(file);
                       } else {
                         const reader = new FileReader();
                         reader.onload = (ev) => { field.set(ev.target.result); showNotification("Ficheiro carregado!", "success"); };
                         reader.readAsText(file);
                       }
                    }} />
                  </label>
                </div>
                <textarea className="flex-1 w-full p-6 border border-slate-100 rounded-2xl bg-slate-50/30 focus:bg-white focus:ring-4 focus:ring-indigo-500/5 outline-none resize-none font-mono text-xs leading-relaxed transition-all" placeholder={field.ph} value={field.val} onChange={(e) => field.set(e.target.value)} />
              </div>
            ))}
            <div className="md:col-span-2 flex justify-end">
              <button onClick={() => setActiveTab('template')} className="flex items-center gap-3 px-10 py-5 rounded-2xl font-black text-xs uppercase tracking-widest text-slate-700 bg-white border border-slate-200 hover:border-indigo-400 hover:text-indigo-600 transition-all shadow-lg">Definir Estrutura <ArrowRight size={18} /></button>
            </div>
          </div>
        )}

        {activeTab === 'template' && (
          <div className="animate-fadeIn space-y-8">
            <div className="bg-slate-50/80 p-6 rounded-3xl border border-slate-200 shadow-inner flex flex-col md:flex-row gap-5 items-end">
              <div className="flex-1 w-full">
                <label className="block text-[10px] font-black text-slate-400 mb-2 uppercase ml-2 tracking-widest">Biblioteca de Modelos</label>
                <select value={selectedTemplateId} onChange={(e) => { const t = savedTemplates.find(tem => tem.id === e.target.value); if (t) { setSelectedTemplateId(t.id); setTemplate(t.content); } else { setSelectedTemplateId(''); setTemplate(''); } }} className="w-full p-4 border border-slate-300 rounded-2xl text-xs bg-white font-bold outline-none shadow-sm focus:ring-2 focus:ring-indigo-500">
                  <option value="">-- Carregar Modelo --</option>
                  {savedTemplates.map(t => <option key={t.id} value={t.id}>{t.name}</option>)}
                </select>
              </div>
              <div className="flex-1 w-full">
                <label className="block text-[10px] font-black text-slate-400 mb-2 uppercase ml-2 tracking-widest">Nome do Modelo</label>
                <input type="text" value={templateNameInput} onChange={(e) => setTemplateNameInput(e.target.value)} placeholder="Ex: Geral" className="w-full p-4 border border-slate-300 rounded-2xl text-xs font-bold outline-none shadow-sm focus:ring-2 focus:ring-indigo-500" />
              </div>
              <div className="flex gap-3">
                <button onClick={async () => {
                  if (!user || !template.trim()) return;
                  try {
                    if (selectedTemplateId) {
                      await updateDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'templates', selectedTemplateId), { content: template, updatedAt: Date.now() });
                    } else {
                      if (!templateNameInput.trim()) return showNotification("Dê um nome!", "error");
                      await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'templates'), { name: templateNameInput, content: template, createdAt: Date.now() });
                      setTemplateNameInput('');
                    }
                    showNotification("Guardado!", "success");
                  } catch (e) { showNotification("Erro ao salvar.", "error"); }
                } } className="bg-green-600 text-white px-6 py-4 rounded-2xl text-[10px] font-black uppercase tracking-widest flex items-center gap-2 shadow-lg hover:bg-green-700 transition-all"><Save size={18}/> Salvar</button>
                <button onClick={() => {setSelectedTemplateId(''); setTemplate(''); setTemplateNameInput('');}} className="bg-slate-200 text-slate-700 px-6 py-4 rounded-2xl text-[10px] font-black uppercase tracking-widest flex items-center gap-2 hover:bg-slate-300 transition-all"><Plus size={18}/> Novo</button>
              </div>
            </div>

            <div className="grid md:grid-cols-3 gap-8 h-[550px]">
              <div className="md:col-span-2 h-full">
                <textarea 
                  className="w-full h-full p-8 border border-slate-200 rounded-[2rem] bg-white font-mono text-xs leading-relaxed focus:ring-8 focus:ring-indigo-500/5 outline-none transition-all shadow-sm" 
                  value={template} 
                  onChange={(e) => setTemplate(e.target.value)} 
                  placeholder={`RELATÓRIO CLÍNICO\nNOME: {{ NOME }}\n...`} 
                />
              </div>
              
              <div className="bg-white rounded-[2rem] border border-slate-200 shadow-sm flex flex-col h-full overflow-hidden">
                <div className="p-6 pb-2 border-b border-slate-50">
                  <h3 className="font-black text-[10px] text-slate-400 uppercase tracking-widest text-center">Campos Detectados</h3>
                </div>
                
                <div className="flex-1 overflow-y-auto p-6 space-y-3 custom-scrollbar">
                  {detectedVars.map((v, i) => (
                    <div key={i} className="text-[10px] bg-slate-50 p-4 rounded-2xl border border-slate-100 font-mono text-indigo-700 font-black flex items-center gap-3 shadow-inner">
                      <CheckCircle size={14} className="text-teal-500 flex-shrink-0"/> 
                      <span className="truncate">{v}</span>
                    </div>
                  ))}
                  {detectedVars.length === 0 && (
                    <div className="text-center py-20 opacity-30">
                      <List className="mx-auto mb-3 text-slate-300" size={32}/>
                      <p className="text-[10px] font-black uppercase leading-tight text-slate-400">Use tags como<br/><span className="text-indigo-600">{"{{"} VAR {"}}"}</span></p>
                    </div>
                  )}
                </div>
                
                <div className="p-6 border-t border-slate-50">
                  <button onClick={() => setActiveTab('result')} className="w-full py-4 rounded-2xl font-black text-[10px] uppercase tracking-[0.2em] bg-indigo-600 text-white flex items-center justify-center gap-3 hover:bg-indigo-700 shadow-2xl transition-all active:scale-95">Ir para Relatório <ArrowRight size={16}/></button>
                </div>
              </div>
            </div>
          </div>
        )}

        {activeTab === 'result' && (
          <div className="animate-fadeIn pb-32">
            {isGenerating ? (
              <div className="h-[600px] flex flex-col items-center justify-center bg-white rounded-[3rem] border border-slate-200 shadow-xl animate-pulse">
                <div className="relative mb-10">
                  <BrainCircuit size={80} className="text-indigo-500 animate-bounce" />
                  <Sparkles size={32} className="text-teal-400 absolute -top-4 -right-4 animate-ping" />
                </div>
                <h3 className="text-2xl font-black text-indigo-900 uppercase tracking-widest">IA em processamento...</h3>
                <p className="text-slate-400 font-bold mt-2 uppercase text-[10px] tracking-[0.3em]">{openaiModel}</p>
              </div>
            ) : !result ? (
              <div className="h-[500px] flex flex-col items-center justify-center bg-white rounded-[3rem] border border-slate-200 text-center p-12 shadow-inner">
                <Sparkles size={72} className="text-indigo-100 mb-10" />
                <h2 className="text-3xl font-black mb-4 uppercase tracking-tighter text-slate-800">Pronto para Consolidar</h2>
                <button onClick={generateReport} className="px-16 py-6 text-xl bg-indigo-600 hover:bg-indigo-700 text-white rounded-[2rem] font-black uppercase tracking-widest flex items-center gap-4 transition-all shadow-2xl">Gerar com {openaiModel}</button>
              </div>
            ) : (
              <div className="grid md:grid-cols-4 gap-10">
                <div className="md:col-span-3 space-y-10">
                  <div className="bg-white rounded-[3rem] shadow-2xl border border-slate-200 overflow-hidden">
                    <div className="bg-slate-50/80 p-5 flex justify-between items-center border-b border-slate-100">
                      <div className="flex items-center gap-4 ml-6">
                        <span className="text-[10px] font-black uppercase text-slate-400 tracking-[0.4em]">Relatório Final</span>
                        {usageCost && (
                          <div className="flex items-center gap-2 bg-indigo-100/50 px-3 py-1 rounded-full text-indigo-700 border border-indigo-200/50">
                            <Coins size={12} className="animate-pulse" />
                            <span className="text-[10px] font-black tracking-tight uppercase">
                              Custo Est.: R$ {usageCost.brl.replace('.', ',')}
                            </span>
                          </div>
                        )}
                      </div>
                      <div className="flex gap-3 mr-4">
                        <button onClick={() => copyToClipboard(result)} className="p-3 hover:bg-white rounded-2xl text-slate-500 border border-transparent hover:border-slate-200 transition-all hover:shadow-md" title="Copiar"><Copy size={20}/></button>
                        <button onClick={() => {
                          const element = document.createElement("a");
                          const file = new Blob([result], {type: 'text/plain'});
                          element.href = URL.createObjectURL(file);
                          element.download = "relatorio.txt";
                          element.click();
                        }} className="p-3 hover:bg-white rounded-2xl text-slate-500 border border-transparent hover:border-slate-200 transition-all hover:shadow-md" title="Descarregar"><Download size={20}/></button>
                      </div>
                    </div>
                    <div className="p-16 font-serif whitespace-pre-wrap leading-[2] text-xl text-slate-900 bg-[radial-gradient(#f1f5f9_1.5px,transparent_1.5px)] [background-size:32px_32px] min-h-[600px] shadow-inner">{result}</div>
                  </div>
                  {extraContent && (
                    <div className="bg-indigo-950 text-white rounded-[3rem] p-12 shadow-2xl animate-fadeIn border-4 border-indigo-800/30">
                      <div className="flex justify-between items-center mb-10">
                        <h4 className="font-black text-xs uppercase tracking-[0.4em] text-indigo-400 flex items-center gap-3">{extraContent.type === 'patient' ? <MessageCircle size={20}/> : <HelpCircle size={20}/>}{extraContent.type === 'patient' ? "Versão para Paciente" : "Seguimento Sugerido"}</h4>
                        <button onClick={() => setExtraContent(null)} className="text-white/20 hover:text-white p-2"><X size={24}/></button>
                      </div>
                      <div className="text-lg leading-[1.8] whitespace-pre-wrap font-medium text-indigo-50">{extraContent.text}</div>
                    </div>
                  )}
                </div>
                <div className="space-y-8">
                  <div className="bg-white p-8 rounded-[2.5rem] border-2 border-slate-100 shadow-xl sticky top-8 text-center">
                    <h4 className="text-[10px] font-black uppercase text-slate-400 tracking-[0.2em] mb-6">Ações</h4>
                    <button onClick={generateReport} className="w-full py-5 bg-indigo-600 hover:bg-indigo-700 text-white rounded-2xl font-black text-[10px] uppercase tracking-widest flex items-center justify-center gap-3 shadow-xl transition-all"><RefreshCw size={16}/> Regenerar</button>
                    <div className="mt-6 space-y-4">
                      <button onClick={() => handleExtraAnalysis('patient')} disabled={isExtraLoading} className="w-full py-4 text-[10px] font-black uppercase bg-slate-50 border-2 border-slate-200 text-slate-700 rounded-2xl flex items-center justify-center gap-3 hover:bg-indigo-50 transition-all">{isExtraLoading ? "..." : "Modo Paciente"}</button>
                      <button onClick={() => handleExtraAnalysis('clinic')} disabled={isExtraLoading} className="w-full py-4 text-[10px] font-black uppercase bg-slate-50 border-2 border-slate-200 text-slate-700 rounded-2xl flex items-center justify-center gap-3 hover:bg-indigo-50 transition-all">{isExtraLoading ? "..." : "Follow-up"}</button>
                    </div>
                  </div>
                </div>
              </div>
            )}
          </div>
        )}
      </main>

      {notification && (
        <div className={`fixed bottom-10 right-10 px-8 py-5 rounded-[2rem] shadow-2xl text-white flex items-center gap-4 animate-bounce-in z-[100] border-2 border-white/20 backdrop-blur-xl ${notification.type === 'success' ? 'bg-teal-600' : 'bg-rose-600'}`}>
          {notification.type === 'success' ? <CheckCircle size={24} /> : <AlertCircle size={24} />}
          <span className="text-[11px] font-black uppercase tracking-widest">{notification.msg}</span>
        </div>
      )}
    </div>
  );
};

export default App;
